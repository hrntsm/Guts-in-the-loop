cmake_minimum_required(VERSION 3.16)
project(SequentialGalleryNative LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POLICY_VERSION_MINIMUM 3.5)
set(ZIN_BUILD_APP OFF CACHE BOOL "" FORCE)
set(ZIN_BUILD_TESTS OFF CACHE BOOL "" FORCE)
# Eigen is header-only. Keep dependency installation inside the checkout.
add_library(Eigen3::Eigen INTERFACE IMPORTED GLOBAL)
set_target_properties(Eigen3::Eigen PROPERTIES INTERFACE_INCLUDE_DIRECTORIES "${CMAKE_CURRENT_SOURCE_DIR}/../../.cache/third_party/eigen")
set(Eigen3_DIR "${CMAKE_CURRENT_SOURCE_DIR}/cmake" CACHE PATH "" FORCE)
add_subdirectory(../../.cache/third_party/sequential-gallery upstream)
add_library(SequentialGallery.Native SHARED bridge.cpp)
target_link_libraries(SequentialGallery.Native PRIVATE sps)
target_compile_definitions(SequentialGallery.Native PRIVATE NOMINMAX _USE_MATH_DEFINES)
set_target_properties(SequentialGallery.Native PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/out")
add_executable(SequentialGallery.WindowSmoke ../../tests/NativeSmoke/window-smoke.cpp)
target_include_directories(SequentialGallery.WindowSmoke PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(SequentialGallery.WindowSmoke PRIVATE sps)
target_compile_definitions(SequentialGallery.WindowSmoke PRIVATE NOMINMAX _USE_MATH_DEFINES)
set_target_properties(SequentialGallery.WindowSmoke PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/out")
